// KEY COLUMN: dim_occupation[occupation_nesting]
//   "top"    = 5 major occupation categories (use for all charts)
//   "branch" = sub-categories (not yet pulled)
//   "leaf"   = detail occupations (not yet pulled)
//   Always filter to "top" to avoid double-counting parent + child
//   rows when occupation_label is on a visual axis.
//
// RACE SUPPORT:
//   117 race-iterated rows are in dim_occupation[race].
//   The by-race measures are identical to the sex-only versions —
//   race breakdown comes from putting dim_occupation[race] on a
//   visual axis or legend, NOT from a hardcoded race filter in DAX.

// ============================================================
// SECTION 1 — CORE WORKER COUNTS
// ============================================================

[Occupation Worker Count Female] =
CALCULATE(
    SUM(fact2_acs_wi_county[estimate]),
    dim_occupation[table] = "C24010",
    dim_occupation[sex] = "Female",
    dim_occupation[occupation_nesting] = "top"
)

[Occupation Worker Count Male] =
CALCULATE(
    SUM(fact2_acs_wi_county[estimate]),
    dim_occupation[table] = "C24010",
    dim_occupation[sex] = "Male",
    dim_occupation[occupation_nesting] = "top"
)

[Occupation Worker Count Total] =
[Occupation Worker Count Male] + [Occupation Worker Count Female]

[Female Share of Occupation Workers] =
DIVIDE(
    [Occupation Worker Count Female],
    [Occupation Worker Count Total]
)


// ============================================================
// SECTION 2 — GAP MEASURES
// ============================================================

[Occupation Worker Count Gap (F minus M)] =
[Occupation Worker Count Female] - [Occupation Worker Count Male]

[Occupation Worker Count Gap Label] =
VAR gap = [Occupation Worker Count Gap (F minus M)]
RETURN
IF(
    gap > 0,
    "There are " & FORMAT(ABS(gap), "#,##0") & " more women than men in this occupation category",
    "There are " & FORMAT(ABS(gap), "#,##0") & " more men than women in this occupation category"
)


// ============================================================
// SECTION 3 — STATEWIDE KPI VERSIONS
// ============================================================

[Occupation Worker Count Female — Statewide] =
CALCULATE(
    [Occupation Worker Count Female],
    ALL(dim_geography)
)

[Occupation Worker Count Male — Statewide] =
CALCULATE(
    [Occupation Worker Count Male],
    ALL(dim_geography)
)

[Occupation Worker Count Gap — Statewide] =
[Occupation Worker Count Female — Statewide] -
[Occupation Worker Count Male — Statewide]

[Female Share of Occupation Workers — Statewide] =
CALCULATE(
    [Female Share of Occupation Workers],
    ALL(dim_geography)
)


// ============================================================
// SECTION 4 — BY RACE
// Identical DAX to sex-only versions — race breakdown comes
// from adding dim_occupation[race] to a visual axis or legend.
// Do NOT hardcode a race filter here.
// Supports 5 major categories only (occupation_nesting = "top")
// ============================================================

[Occupation Worker Count Female by Race] =
CALCULATE(
    SUM(fact2_acs_wi_county[estimate]),
    dim_occupation[table] = "C24010",
    dim_occupation[sex] = "Female",
    dim_occupation[occupation_nesting] = "top"
)

[Occupation Worker Count Male by Race] =
CALCULATE(
    SUM(fact2_acs_wi_county[estimate]),
    dim_occupation[table] = "C24010",
    dim_occupation[sex] = "Male",
    dim_occupation[occupation_nesting] = "top"
)

[Female Share of Occupation Workers by Race] =
DIVIDE(
    [Occupation Worker Count Female by Race],
    [Occupation Worker Count Female by Race] + [Occupation Worker Count Male by Race]
)
